<form-maxlength-progress> via esm.sh

README and API documentation

Class-only import and manual registration

This page loads the component class from esm.sh and registers it manually. The labeled field works without JavaScript or if the CDN fails; native maxlength still applies. 

<label for="esm-message">Message</label>
<textarea id="esm-message" maxlength="100"></textarea>
<form-maxlength-progress for="esm-message">
</form-maxlength-progress>

import { FormMaxlengthProgressElement } from
  'https://esm.sh/@aarongustafson/form-maxlength-progress@1.1.3/form-maxlength-progress.js?bundle&target=es2022';

if (!customElements.get('form-maxlength-progress')) {
  customElements.define(
    'form-maxlength-progress',
    FormMaxlengthProgressElement,
  );
}
